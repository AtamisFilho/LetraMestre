   238→  const [starting, setStarting] = useState(false);
   239→
   240→  useEffect(() => {

   242→      onPlayerList: (players: any) => useGameStore.getState().setPlayerList(players),
   243→      onGameStarted: (state: any) => {
   244→        useGameStore.getState().setGameState(state);
   245→        useGameStore.getState().setScreen('game');
   246→      },
   247→      onChatMessage: (data: any) => {
   248→        useGameStore.getState().addChatMessage(data);

   256→    if (!gameId || !playerId) return;
   257→    setStarting(true);
   258→    try {
   259→      const result = await startGame(gameId, playerId);
   260→      if (result.success) {
   261→        toast.success('Jogo iniciado!');
   262→      } else {
   263→        toast.error(result.error || 'Erro ao iniciar');

   265→    } catch (err) {
   266→      toast.error('Erro ao iniciar jogo');

   268→      setStarting(false);
   269→    }
   270→  };

   288→            <div className="flex items-center justify-center gap-2">
   289→              <span className="text-3xl font-mono font-bold tracking-widest">{gameCode}</span>
   290→              <Button size="sm" variant="ghost" onClick={handleCopyCode}>
   291→                <Copy className="h-4 w-4" />

   293→            </div>
   294→            <p className="text-xs text-muted-foreground mt-1">Compartilhe este código com outros jogadores</p>

   297→          {/* Player List */}

   299→            <p className="text-sm font-medium">Jogadores ({playerList.length}/4)</p>
   300→            {playerList.map((player) => (
   301→              <div key={player.id} className="flex items-center gap-3 p-3 bg-muted/50 rounded-lg">

   314→              onClick={handleStart}

   325→            </div>

   327→
   328→          <Button variant="ghost" onClick={() => { disconnectSocket(); setScreen('home'); }} className="w-full text-destructive">
   329→            Sair da Sala
   330→          </Button>
   331→        </CardContent>
   332→      </Card>
   333→    </div>

   337→// ─── Game Screen ───────────────────────────────────────────────────────────────
   338→export function GameScreen() {
   339→  const store = useGameStore();
   340→  const {
   341→    gameState, playerId, isHost, gameId, gameCode, selectedTileIndices,
   342→    toggleTileSelection, clearSelection, placedTiles, addPlacedTile, 
   343→    removePlacedTile, clearPlacedTiles, placingTileIndex, setPlacingTileIndex,
   344→    pendingWord, setPendingWord, gameOverData, setGameOverData, 
   345→    addChatMessage, setScreen, setError, lastError
   346→  } = store;
   347→
   348→  const [showChat, setShowChat] = useState(false);
   349→  const [showExchange, setShowExchange] = useState(false);
   350→  const [exchangeIndices, setExchangeIndices] = useState<Set<number>>(new Set());
   351→
   352→  // Setup socket listeners
   353→  useEffect(() => {
   354→    const callbacks = {
   355→      onGameState: (state: GameState) => {
   356→        useGameStore.getState().setGameState(state);
   357→        // Clear placed tiles when state updates (move was accepted)
   358→        useGameStore.getState().clearPlacedTiles();
   359→        useGameStore.getState().clearSelection();
   360→      },
   361→      onGameEnded: (data: any) => {
   362→        useGameStore.getState().setGameOverData(data);
   363→      },
   364→      onWordValidation: (data: any) => {
   365→        useGameStore.getState().setPendingWord(data.word);
   366→      },
   367→      onChatMessage: (data: any) => {
   368→        useGameStore.getState().addChatMessage(data);
   369→      },
   370→    };
   371→    setupGameListeners(callbacks);
   372→    return () => removeGameListeners();
   373→  }, []);
   374→
   375→  if (!gameState || !playerId) return null;
   376→
   377→  const myPlayer = gameState.players.find(p => p.id === playerId);
   378→  const currentPlayer = gameState.players[gameState.currentPlayerIndex];
   379→  const isMyTurn = currentPlayer?.id === playerId;
   380→
   381→  // Handle tile click in rack
   382→  const handleTileClick = (index: number) => {
   383→    if (!isMyTurn) return;
   384→    // If no tile is being placed, select for placement
   385→    if (placingTileIndex === null) {
   386→      // Toggle selection
   387→      if (selectedTileIndices.has(index)) {
   388→        toggleTileSelection(index);
   389→      } else {
   390→        setPlacingTileIndex(index);
   391→        toggleTileSelection(index);
   392→      }
   393→    } else {
   394→      // Already placing a tile - switch to this one
   395→      setPlacingTileIndex(index);
   396→      clearSelection();
   397→      toggleTileSelection(index);
   398→    }
   399→  };
   400→
   401→  // Handle cell click on board
   402→  const handleCellClick = (row: number, col: number) => {
   403→    if (!isMyTurn || !myPlayer) return;
   404→    
   405→    // Check if there's a placed tile at this position - remove it
   406→    const key = `${row},${col}`;
   407→    if (placedTiles.has(key)) {
   408→      removePlacedTile(row, col);
   409→      return;
   410→    }
   411→
   412→    // If we have a tile selected for placing
   413→    if (placingTileIndex !== null && placingTileIndex < myPlayer.rack.length) {
   414→      const tile = myPlayer.rack[placingTileIndex];
   415→      const placement: TilePlacement = {
   416→        row, col, tile, rackIndex: placingTileIndex,
   417→      };
   418→      addPlacedTile(placement);
   419→      setPlacingTileIndex(null);
   420→    }
   421→  };
   422→
   423→  // Confirm move
   424→  const handleConfirmMove = async () => {
   425→    if (!gameId || !playerId || placedTiles.size === 0) return;
   426→    
   427→    const placements = Array.from(placedTiles.values());
   428→    const moveData: MoveData = {
   429→      playerId,
   430→      placements,
   431→    };
   432→
   433→    try {
   434→      const result = await submitMove(gameId, moveData);
   435→      if (result.success) {
   436→        clearPlacedTiles();
   437→        clearSelection();
   438→        toast.success(`Jogada válida! +${result.score} pontos (${result.words?.join(', ')})`);
   439→      } else {
   440→        toast.error(result.error || 'Jogada inválida');
   441→      }
   442→    } catch (err) {
   443→      toast.error('Erro ao enviar jogada');
   444→    }
   445→  };
   446→
   447→  // Pass turn
   448→  const handlePassTurn = async () => {
   449→    if (!gameId || !playerId) return;
   450→    clearPlacedTiles();
   451→    clearSelection();
   452→    try {
   453→      await passTurn(gameId, playerId);
   454→      toast.info('Você passou a vez');
   455→    } catch (err) {
   456→      toast.error('Erro ao passar');
   457→    }
   458→  };
   459→
   460→  // Exchange tiles
   461→  const handleExchange = async () => {
   462→    if (!gameId || !playerId || exchangeIndices.size === 0) return;
   463→    try {
   464→      await exchangeTiles(gameId, playerId, Array.from(exchangeIndices));
   465→      setExchangeIndices(new Set());
   466→      setShowExchange(false);
   467→      toast.info('Peças trocadas!');
   468→    } catch (err) {
   469→      toast.error('Erro ao trocar peças');
   470→    }
   471→  };
   472→
   473→  // Word validation
   474→  const handleWordApproval = async (approved: boolean) => {
   475→    if (!gameId || !playerId) return;
   476→    try {
   477→      await approveWord(gameId, playerId, approved);
   478→      setPendingWord(null);
   479→      toast.info(approved ? 'Palavra aprovada!' : 'Palavra rejeitada');
   480→    } catch (err) {
   481→      toast.error('Erro ao validar palavra');
   482→    }
   483→  };
   484→
   485→  // Build display board with placed tiles
   486→  let displayBoard = gameState.board;
   487→  for (const placement of placedTiles.values()) {
   488→    displayBoard = withTilePlaced(displayBoard, placement.row, placement.col, placement.tile, true);
   489→  }
   490→
   491→  return (
   492→    <div className="min-h-screen flex flex-col bg-gradient-to-b from-amber-50 to-amber-100">
   493→      {/* Top bar */}
   494→      <div className="bg-primary text-primary-foreground px-3 py-2 flex items-center justify-between">
   495→        <div className="flex items-center gap-2">
   496→          <Gamepad2 size={18} />
   497→          <span className="font-bold text-sm">LetraMestre</span>
   498→          <Badge variant="secondary" className="text-xs">{gameCode}</Badge>
   499→        </div>
   500→        <div className="flex items-center gap-2">
   501→          <Button size="sm" variant="ghost" className="text-primary-foreground h-7 w-7 p-0" onClick={() => setShowChat(!showChat)}>
   502→            <MessageSquare size={16} />
   503→          </Button>
   504→          <Button size="sm" variant="ghost" className="text-primary-foreground h-7 w-7 p-0" onClick={() => { disconnectSocket(); setScreen('home'); }}>
   505→            <Home size={16} />
   506→          </Button>
   507→        </div>
   508→      </div>
   509→
   510→      {/* Score board */}
   511→      <ScoreBoard players={gameState.players} currentPlayerId={currentPlayer?.id} myPlayerId={playerId} />
   512→
   513→      {/* Turn indicator */}
   514→      <div className={`px-3 py-1.5 text-center text-sm font-medium ${isMyTurn ? 'bg-amber-500 text-white' : 'bg-muted text-muted-foreground'}`}>
   515→        {isMyTurn ? (
   516→          <span className="flex items-center justify-center gap-1">
   517→            <Sparkles size={14} /> Sua vez de jogar!
   518→          </span>
   519→        ) : (
   520→          <span>Vez de {currentPlayer?.name ?? ''}</span>
   521→        )}
   522→      </div>
   523→
   524→      {/* Board */}
   525→      <div className="flex-1 flex items-center justify-center p-1 overflow-auto">
   526→        <BoardView board={displayBoard} onCellClick={handleCellClick} isMyTurn={isMyTurn} />
   527→      </div>
   528→
   529→      {/* Tile bag info */}
   530→      <div className="text-center text-xs text-muted-foreground py-1">
   531→        Peças no saco: {gameState.tileBagCount}
   532→      </div>
   533→
   534→      {/* Player's rack */}
   535→      <div className="px-2 pb-1">
   536→        <RackView
   537→          tiles={myPlayer?.rack || []}
   538→          selectedIndices={selectedTileIndices}
   539→          onTileClick={handleTileClick}
   540→          disabled={!isMyTurn}
   541→        />
   542→      </div>
   543→
   544→      {/* Action buttons */}
   545→      {isMyTurn && (
   546→        <div className="flex gap-2 px-3 pb-3">
   547→          <Button
   548→            variant="outline"
   549→            size="sm"
   550→            className="flex-1 h-10"
   551→            onClick={handlePassTurn}
   552→          >
   553→            <Hand size={16} className="mr-1" /> Passar
   554→          </Button>
   555→          <Button
   556→            variant="outline"
   557→            size="sm"
   558→            className="flex-1 h-10"
   559→            onClick={() => setShowExchange(true)}
   560→            disabled={gameState.tileBagCount < 7}
   561→          >
   562→            <Shuffle size={16} className="mr-1" /> Trocar
   563→          </Button>
   564→          <Button
   565→            size="sm"
   566→            className="flex-1 h-10 bg-amber-500 hover:bg-amber-600 text-white"
   567→            onClick={handleConfirmMove}
   568→            disabled={placedTiles.size === 0}
   569→          >
   570→            <Check size={16} className="mr-1" /> Confirmar
   571→          </Button>
   572→        </div>
   573→      )}
   574→
   575→      {/* Last move info */}
   576→      {gameState.lastPlayedWord && (
   577→        <div className="text-center text-xs text-muted-foreground pb-2">
   578→          Última jogada: &quot;{gameState.lastPlayedWord}&quot; (+{gameState.lastPlayedScore} pts)
   579→        </div>
   580→      )}
   581→
   582→      {/* Chat drawer */}
   583→      <Dialog open={showChat} onOpenChange={setShowChat}>
   584→        <DialogContent className="max-w-sm h-[60vh] flex flex-col p-0">
   585→          <DialogHeader className="p-4 pb-0">
   586→            <DialogTitle className="flex items-center gap-2">
   587→              <MessageSquare size={18} /> Chat
   588→            </DialogTitle>
   589→          </DialogHeader>
   590→          <ChatPanel />
   591→        </DialogContent>
   592→      </Dialog>
   593→
   594→      {/* Exchange dialog */}
   595→      <Dialog open={showExchange} onOpenChange={setShowExchange}>
   596→        <DialogContent className="max-w-sm">
   597→          <DialogHeader>
   598→            <DialogTitle>Trocar Peças</DialogTitle>
   599→            <DialogDescription>Selecione as peças que deseja trocar</DialogDescription>
   600→          </DialogHeader>
   601→          <div className="game-rack flex items-center justify-center gap-1 flex-wrap">
   602→            {(myPlayer?.rack || []).map((tile, i) => (
   603→              <button
   604→                key={i}
   605→                className={`tile-piece ${exchangeIndices.has(i) ? 'selected' : ''}`}
   606→                style={{ width: 44, height: 44, fontSize: 20 }}
   607→                onClick={() => {
   608→                  const next = new Set(exchangeIndices);
   609→                  if (next.has(i)) next.delete(i); else next.add(i);
   610→                  setExchangeIndices(next);
   611→                }}
   612→              >
   613→                {tile.isBlank ? '?' : tile.letter}
   614→                {tile.value > 0 && <span className="tile-value">{tile.value}</span>}
   615→              </button>
   616→            ))}
   617→          </div>
   618→          <DialogFooter>
   619→            <Button variant="outline" onClick={() => { setShowExchange(false); setExchangeIndices(new Set()); }}>Cancelar</Button>
   620→            <Button onClick={handleExchange} disabled={exchangeIndices.size === 0} className="bg-amber-500 hover:bg-amber-600 text-white">
   621→              Trocar {exchangeIndices.size} peça(s)
   622→            </Button>
   623→          </DialogFooter>
   624→        </DialogContent>
   625→      </Dialog>
   626→
   627→      {/* Word validation dialog */}
   628→      <Dialog open={!!pendingWord} onOpenChange={() => setPendingWord(null)}>
   629→        <DialogContent className="max-w-sm">
   630→          <DialogHeader>
   631→            <DialogTitle>Validar Palavra</DialogTitle>
   632→            <DialogDescription>Esta palavra não foi encontrada no dicionário</DialogDescription>
   633→          </DialogHeader>
   634→          <div className="text-center py-4">
   635→            <p className="text-4xl font-bold text-primary mb-2">{pendingWord}</p>
   636→            <p className="text-sm text-muted-foreground">Os jogadores podem validar esta palavra manualmente.</p>
   637→          </div>
   638→          <DialogFooter className="flex gap-2">
   639→            <Button variant="outline" onClick={() => handleWordApproval(false)} className="flex-1">
   640→              <X size={16} className="mr-1" /> Rejeitar
   641→            </Button>
   642→            <Button onClick={() => handleWordApproval(true)} className="flex-1 bg-green-600 hover:bg-green-700 text-white">
   643→              <Check size={16} className="mr-1" /> Aprovar
   644→            </Button>
   645→          </DialogFooter>
   646→        </DialogContent>
   647→      </Dialog>
   648→
   649→      {/* Game over dialog */}
   650→      <Dialog open={!!gameOverData} onOpenChange={() => setGameOverData(null)}>
   651→        <DialogContent className="max-w-sm">
   652→          <DialogHeader>
   653→            <DialogTitle className="text-center">🏆 Fim de Jogo!</DialogTitle>
   654→          </DialogHeader>
   655→          <div className="text-center py-4 space-y-4">
   656→            <div>
   657→              <p className="text-sm text-muted-foreground">Vencedor</p>
   658→              <p className="text-3xl font-bold text-primary">{gameOverData?.winnerName}</p>
   659→            </div>
   660→            <div className="space-y-1">
   661→              {gameOverData?.scores && Object.entries(gameOverData.scores).map(([name, score]) => (
   662→                <div key={name} className="flex justify-between px-4 py-1 bg-muted rounded">
   663→                  <span className="font-medium">{name}</span>
   664→                  <span className="font-bold">{score} pts</span>
   665→                </div>
   666→              ))}
   667→            </div>
   668→          </div>
   669→          <DialogFooter>
   670→            <Button onClick={() => { disconnectSocket(); setScreen('home'); }} className="w-full bg-amber-500 hover:bg-amber-600 text-white">
   671→              <Home size={16} className="mr-2" /> Voltar ao Início
   672→            </Button>
   673→          </DialogFooter>
   674→        </DialogContent>
   675→      </Dialog>
   676→    </div>
   677→  );
   678→}
